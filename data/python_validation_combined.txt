def square(n):
    return n * n

def find_max(a, b):
    if a > b:
        return a
    return b

def find_min(a, b):
    if a < b:
        return a
    return b

def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

def is_positive(n):
    return n > 0

def remove_duplicates(items):
    return list(set(items))

def calculate_area(length, width):
    return length * width

def remainder(a, b):
    return a % b

def find_longest_word(sentence):
    words = sentence.split()
    return max(words, key=len)

def rotate_list(items, positions):
    if not items:
        return items
    positions %= len(items)
    return items[-positions:] + items[:-positions]

def second_largest(numbers):
    unique_numbers = sorted(set(numbers))
    if len(unique_numbers) < 2:
        return None
    return unique_numbers[-2]

def binary_search(items, target):
    low = 0
    high = len(items) - 1
    while low <= high:
        middle = (low + high) // 2
        if items[middle] == target:
            return middle
        if items[middle] < target:
            low = middle + 1
        else:
            high = middle - 1
    return -1

def count_occurrences(items, target):
    return items.count(target)

def matrix_diagonal_sum(matrix):
    return sum(matrix[i][i] for i in range(min(len(matrix), len(matrix[0]))))

def power_recursive(base, exponent):
    if exponent == 0:
        return 1
    if exponent < 0:
        return 1 / power_recursive(base, -exponent)
    return base * power_recursive(base, exponent - 1)

def calculate_mean(numbers):
    if not numbers:
        return None
    return sum(numbers) / len(numbers)

def calculate_median(numbers):
    if not numbers:
        return None
    values = sorted(numbers)
    middle = len(values) // 2
    if len(values) % 2 == 0:
        return (values[middle - 1] + values[middle]) / 2
    return values[middle]

def unzip_pairs(pairs):
    if not pairs:
        return [], []
    first, second = zip(*pairs)
    return list(first), list(second)

def get_tuple_first(items):
    return items[0] if items else None

def remove_list_element(items, value):
    result = items.copy()
    if value in result:
        result.remove(value)
    return result

def split_even_odd(numbers):
    even = [n for n in numbers if n % 2 == 0]
    odd = [n for n in numbers if n % 2 != 0]
    return even, odd

def get_initials(full_name):
    return "".join(word[0].upper() for word in full_name.split() if word)

def dictionary_contains_key(data, key):
    return key in data

def remove_dictionary_key(data, key):
    result = data.copy()
    result.pop(key, None)
    return result

def get_common_elements(first, second):
    return list(set(first) & set(second))

def calculate_dot_product(first, second):
    return sum(a * b for a, b in zip(first, second))

def find_unique_characters(text):
    return list(dict.fromkeys(text))

def generate_squares(n):
    return [i * i for i in range(n)]

def count_matching_items(items, condition):
    return sum(1 for item in items if condition(item))

def get_top_n(numbers, count):
    return sorted(numbers, reverse=True)[:count]

def calculate_percentage_change(old_value, new_value):
    if old_value == 0:
        return None
    return ((new_value - old_value) / abs(old_value)) * 100

def find_most_frequent_item(items):
    if not items:
        return None
    return max(set(items), key=items.count)

def move_zeros_to_end(numbers):
    nonzero = [n for n in numbers if n != 0]
    return nonzero + [0] * (len(numbers) - len(nonzero))

def remove_adjacent_duplicates(items):
    if not items:
        return []
    result = [items[0]]
    for item in items[1:]:
        if item != result[-1]:
            result.append(item)
    return result

def find_matrix_maximum(matrix):
    values = [value for row in matrix for value in row]
    return max(values) if values else None

def merge_frequency_maps(first, second):
    result = first.copy()
    for key, value in second.items():
        result[key] = result.get(key, 0) + value
    return result

def calculate_distance(x1, y1, x2, y2):
    return ((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5

def find_missing_element(numbers, n):
    return n * (n + 1) // 2 - sum(numbers)

def calculate_prefix_sums(numbers):
    result = []
    total = 0
    for number in numbers:
        total += number
        result.append(total)
    return result

def find_maximum_subarray_sum(numbers):
    if not numbers:
        return None
    current = maximum = numbers[0]
    for number in numbers[1:]:
        current = max(number, current + number)
        maximum = max(maximum, current)
    return maximum

def group_by_remainder(numbers, divisor):
    if divisor == 0:
        return {}
    groups = {}
    for number in numbers:
        groups.setdefault(number % divisor, []).append(number)
    return groups

def find_maximum(numbers):
    if not numbers:
        return None
    return max(numbers)

def count_negative_numbers(numbers):
    return sum(1 for number in numbers if number < 0)

def count_uppercase_letters(text):
    return sum(1 for char in text if char.isupper())

def reverse_each_word(sentence):
    return " ".join(word[::-1] for word in sentence.split())

def calculate_cumulative_sum(numbers):
    result = []
    total = 0
    for number in numbers:
        total += number
        result.append(total)
    return result

def find_common_prefix_length(first, second):
    count = 0
    for a, b in zip(first, second):
        if a != b:
            break
        count += 1
    return count

def remove_empty_strings(items):
    return [item for item in items if item]

def is_valid_index(items, index):
    return -len(items) <= index < len(items)
