
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return None
    return a / b

def square(n):
    return n * n

def cube(n):
    return n * n * n

def is_even(n):
    return n % 2 == 0

def is_odd(n):
    return n % 2 != 0

def greet(name):
    return "Hello " + name

def find_max(a, b):
    if a > b:
        return a
    return b

def find_min(a, b):
    if a < b:
        return a
    return b

def calculate_average(numbers):
    return sum(numbers) / len(numbers)

def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

def reverse_string(text):
    return text[::-1]

def count_items(items):
    return len(items)

def get_first(items):
    return items[0]

def get_last(items):
    return items[-1]

def list_sum(numbers):
    return sum(numbers)

def find_positive(numbers):
    return [n for n in numbers if n > 0]

def count_vowels(text):
    count = 0
    for char in text.lower():
        if char in "aeiou":
            count += 1
    return count

def calculate_percentage(value, total):
    return (value / total) * 100

def is_positive(n):
    return n > 0

def is_negative(n):
    return n < 0

def absolute_value(n):
    if n < 0:
        return -n
    return n

def power(base, exponent):
    return base ** exponent

def convert_to_upper(text):
    return text.upper()

def convert_to_lower(text):
    return text.lower()

def remove_duplicates(items):
    return list(set(items))

def get_second(items):
    return items[1]

def calculate_area(length, width):
    return length * width
