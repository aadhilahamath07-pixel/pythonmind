def frequency_count(items):
    frequency = {}
    for item in items:
        frequency[item] = frequency.get(item, 0) + 1
    return frequency

def mask_string(text, visible):
    if visible < 0:
        visible = 0
    return '*' * max(0, len(text) - visible) + text[-visible:] if visible else '*' * len(text)

def capitalize_words(text):
    return text.title()

def clamp_value(value, minimum, maximum):
    return max(minimum, min(value, maximum))

def calculate_string_similarity(first, second):
    if not first and (not second):
        return 1.0
    if not first or not second:
        return 0.0
    common = sum((1 for a, b in zip(first, second) if a == b))
    return common / max(len(first), len(second))

def set_union(first, second):
    return first | second

def matrix_row_sums(matrix):
    return [sum(row) for row in matrix]

def split_into_groups(items, group_size):
    if group_size <= 0:
        return []
    return [items[i:i + group_size] for i in range(0, len(items), group_size)]

def insertion_sort(items):
    result = items.copy()
    for i in range(1, len(result)):
        current = result[i]
        j = i - 1
        while j >= 0 and result[j] > current:
            result[j + 1] = result[j]
            j -= 1
        result[j + 1] = current
    return result

def get_alphabetically_first(words):
    if not words:
        return None
    return min(words, key=str.lower)

def safe_string_to_float(value):
    try:
        return float(value)
    except (ValueError, TypeError):
        return None

def get_every_second_item(items):
    return items[::2]

def are_anagrams(first, second):
    return sorted(first.lower()) == sorted(second.lower())

def calculate_percentage_change(old_value, new_value):
    if old_value == 0:
        return None
    return (new_value - old_value) / abs(old_value) * 100

def decimal_to_binary(number):
    return bin(number)[2:]

def list_sum(numbers):
    return sum(numbers)

def insert_at_position(items, index, value):
    result = items.copy()
    result.insert(index, value)
    return result

def is_even(n):
    return n % 2 == 0

def dictionary_get_default(data, key, default):
    return data.get(key, default)

def remove_vowels(text):
    return ''.join((char for char in text if char.lower() not in 'aeiou'))

def convert_to_lower(text):
    return text.lower()

def separate_positive_negative(numbers):
    positive = [n for n in numbers if n > 0]
    negative = [n for n in numbers if n < 0]
    return (positive, negative)

def find_missing_dictionary_keys(data, required_keys):
    return [key for key in required_keys if key not in data]

def get_middle_element(items):
    if not items:
        return None
    return items[len(items) // 2]

def create_frequency_dictionary(items):
    result = {}
    for item in items:
        result[item] = result.get(item, 0) + 1
    return result

def filter_dictionary(data, threshold):
    return {key: value for key, value in data.items() if value > threshold}

def safe_get_list_item(items, index):
    try:
        return items[index]
    except IndexError:
        return None

def multiply(a, b):
    return a * b

def get_unique_sorted(items):
    return sorted(set(items))

def find_intersection_count(first, second):
    return len(set(first) & set(second))

def find_last_occurrence(items, target):
    low = 0
    high = len(items) - 1
    result = -1
    while low <= high:
        middle = (low + high) // 2
        if items[middle] == target:
            result = middle
            low = middle + 1
        elif items[middle] < target:
            low = middle + 1
        else:
            high = middle - 1
    return result

def divide(a, b):
    if b == 0:
        return None
    return a / b

def count_vowels(text):
    count = 0
    for char in text.lower():
        if char in 'aeiou':
            count += 1
    return count

def get_bottom_n(numbers, count):
    return sorted(numbers)[:count]

def unique_values(items):
    return list(dict.fromkeys(items))

def calculate_triangle_area(base, height):
    return 0.5 * base * height

def find_second_largest(numbers):
    unique_numbers = sorted(set(numbers))
    if len(unique_numbers) < 2:
        return None
    return unique_numbers[-2]

def find_second_smallest(numbers):
    unique = sorted(set(numbers))
    return unique[1] if len(unique) > 1 else None

def recursive_sum(items):
    if not items:
        return 0
    return items[0] + recursive_sum(items[1:])

def set_difference(first, second):
    return first - second

def format_full_name(first, last):
    return f'{first.strip().title()} {last.strip().title()}'

def generate_even_numbers(n):
    return [i for i in range(n) if i % 2 == 0]

def invert_groups(groups):
    result = {}
    for key, values in groups.items():
        for value in values:
            result.setdefault(value, []).append(key)
    return result
