def count_negative_numbers(numbers):
    return sum(1 for number in numbers if number < 0)


def count_even_numbers(numbers):
    return sum(1 for number in numbers if number % 2 == 0)


def find_smallest_word(words):
    if not words:
        return None
    return min(words, key=len)


def find_longest_word_from_list(words):
    if not words:
        return None
    return max(words, key=len)





def calculate_average_length(words):
    if not words:
        return 0
    return sum(len(word) for word in words) / len(words)


def count_uppercase_letters(text):
    return sum(1 for char in text if char.isupper())


def count_lowercase_letters(text):
    return sum(1 for char in text if char.islower())


def remove_vowels(text):
    return "".join(char for char in text if char.lower() not in "aeiou")


def reverse_each_word(sentence):
    return " ".join(word[::-1] for word in sentence.split())


def find_first_even(numbers):
    for number in numbers:
        if number % 2 == 0:
            return number
    return None


def find_first_negative(numbers):
    for number in numbers:
        if number < 0:
            return number
    return None


def get_middle_element(items):
    if not items:
        return None
    return items[len(items) // 2]


def count_unique_items(items):
    return len(set(items))


def calculate_cumulative_sum(numbers):
    result = []
    total = 0
    for number in numbers:
        total += number
        result.append(total)
    return result


def remove_negative_numbers(numbers):
    return [number for number in numbers if number >= 0]


def get_alphabetically_first(words):
    if not words:
        return None
    return min(words, key=str.lower)


def calculate_difference(numbers):
    if len(numbers) < 2:
        return None
    return max(numbers) - min(numbers)


def find_common_prefix_length(first, second):
    count = 0
    for a, b in zip(first, second):
        if a != b:
            break
        count += 1
    return count


def count_matching_values(first, second):
    return sum(1 for item in first if item in second)


def get_every_second_item(items):
    return items[::2]


def insert_separator(items, separator):
    return separator.join(str(item) for item in items)


def remove_empty_strings(items):
    return [item for item in items if item]


def calculate_factorial_iterative(number):
    if number < 0:
        return None
    result = 1
    for value in range(2, number + 1):
        result *= value
    return result


def is_power_of_two(number):
    return number > 0 and (number & (number - 1)) == 0


def count_divisors(number):
    if number == 0:
        return 0
    return sum(1 for value in range(1, abs(number) + 1)
               if number % value == 0)


def find_missing_number(numbers, n):
    expected = n * (n + 1) // 2
    return expected - sum(numbers)


def rotate_list_right(items, positions):
    if not items:
        return []
    positions %= len(items)
    return items[-positions:] + items[:-positions] if positions else items[:]


def merge_alternating(first, second):
    result = []
    for a, b in zip(first, second):
        result.extend([a, b])
    return result + first[len(second):] + second[len(first):]


def group_numbers_by_sign(numbers):
    positive = [n for n in numbers if n > 0]
    negative = [n for n in numbers if n < 0]
    zero = [n for n in numbers if n == 0]
    return positive, negative, zero


def create_frequency_dictionary(items):
    result = {}
    for item in items:
        result[item] = result.get(item, 0) + 1
    return result


def swap_dictionary_keys_values(data):
    return {value: key for key, value in data.items()}


def get_values_above_average(numbers):
    if not numbers:
        return []
    average = sum(numbers) / len(numbers)
    return [number for number in numbers if number > average]


def calculate_euclidean_distance(x1, y1, x2, y2):
    return ((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5


def is_valid_index(items, index):
    return -len(items) <= index < len(items)


def split_into_equal_parts(items, parts):
    if parts <= 0:
        return []
    size = (len(items) + parts - 1) // parts
    return [items[i:i + size] for i in range(0, len(items), size)]


def remove_common_items(first, second):
    return [item for item in first if item not in second]


def find_second_largest_distinct(numbers):
    unique = sorted(set(numbers))
    return unique[-2] if len(unique) >= 2 else None


def count_words_starting_with(words, letter):
    return sum(1 for word in words if word.lower().startswith(letter.lower()))


def format_phone_number(digits):
    digits = "".join(char for char in str(digits) if char.isdigit())
    return digits


def calculate_weighted_sum(values, weights):
    return sum(value * weight for value, weight in zip(values, weights))
